cmake_minimum_required(VERSION 3.28)

# Locate vcpkg before project(). CMakePresets.json sets these, but CLion will
# happily configure with a profile of its own making; without this that profile
# fails at find_package(EnTT). All binary dirs share one installed tree so a
# second profile does not trigger a second full dependency build.
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    if(DEFINED ENV{VCPKG_ROOT})
        set(_vcpkg_root "$ENV{VCPKG_ROOT}")
    else()
        set(_vcpkg_root "$ENV{HOME}/vcpkg")
    endif()
    if(EXISTS "${_vcpkg_root}/scripts/buildsystems/vcpkg.cmake")
        set(CMAKE_TOOLCHAIN_FILE "${_vcpkg_root}/scripts/buildsystems/vcpkg.cmake"
            CACHE STRING "vcpkg toolchain")
    else()
        message(FATAL_ERROR
            "vcpkg not found at ${_vcpkg_root}. Run tools/install-cpp-deps.sh, "
            "or set VCPKG_ROOT.")
    endif()
endif()

if(NOT DEFINED VCPKG_INSTALLED_DIR)
    set(VCPKG_INSTALLED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/vcpkg_installed"
        CACHE STRING "shared vcpkg install tree")
endif()

# std::ranges::to needs libstdc++ 14; Ubuntu 24.04's default c++ is g++-13 and
# resolves it to nothing. CMakePresets.json pins this too, but CLion configures
# with its own profile that never reads the preset, so pin it here as well.
# Must precede project() — CMake bakes the compiler into the cache at that point
# and ignores every later change to it.
if(NOT DEFINED CMAKE_CXX_COMPILER)
    find_program(_gxx14 NAMES g++-14)
    if(_gxx14)
        set(CMAKE_CXX_COMPILER "${_gxx14}" CACHE FILEPATH "C++ compiler")
    endif()
endif()

project(slime_server CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# No modules here. GCC 14 advertises module-dependency scanning where 13 did not,
# and the -fmodules-ts it adds to every compile line makes CLion's resolver emit
# diagnostics of its own.
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

# Dependencies come from vcpkg.json. See CMakePresets.json for the toolchain.
find_package(EnTT CONFIG REQUIRED)
find_package(flatbuffers CONFIG REQUIRED)
find_package(unofficial-sodium CONFIG REQUIRED)
find_package(unofficial-uwebsockets CONFIG REQUIRED)
find_package(Catch2 3 CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(tomlplusplus CONFIG REQUIRED)

# ----------------------------------------------------------- generated schema
# flatc comes from the same vcpkg port as the C++ headers, so the version match
# that generated code static_asserts on holds by construction. The TypeScript
# client is still generated by ../schema/generate.sh.

# The source root. Every include path and FILE_SET BASE_DIRS below is relative to this, not
# to CMAKE_CURRENT_SOURCE_DIR: sources live under src/ but tests/ does not, so a consumer
# outside src/ only resolves #include "game/world.h" if the -I points at src/ specifically.
set(src_dir "${CMAKE_CURRENT_SOURCE_DIR}/src")
set(schema_dir "${CMAKE_CURRENT_SOURCE_DIR}/../schema")
set(generated_dir "${CMAKE_CURRENT_BINARY_DIR}/generated")

file(GLOB schema_files CONFIGURE_DEPENDS "${schema_dir}/*.fbs")

set(generated_headers "")
foreach(schema IN LISTS schema_files)
    get_filename_component(schema_name "${schema}" NAME_WE)
    list(APPEND generated_headers "${generated_dir}/${schema_name}_generated.h")
endforeach()

add_custom_command(
    OUTPUT ${generated_headers}
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${generated_dir}"
    COMMAND flatbuffers::flatc --cpp --scoped-enums -o "${generated_dir}" ${schema_files}
    DEPENDS ${schema_files} flatbuffers::flatc
    COMMENT "flatc: generating C++ from ${schema_dir}"
    VERBATIM
)

# One target owns running flatc. Without it, the generated headers are attached to whichever
# target happens to list them as sources, and every *other* consumer depends on the header
# with no edge saying who produces it — which ninja resolves as a dependency cycle:
#   common_generated.h -> libslime_world.a -> world.cpp.o -> common_generated.h
add_custom_target(generate_schema DEPENDS ${generated_headers})

add_library(schema INTERFACE)
add_dependencies(schema generate_schema)
# BASE_DIRS and the include directory are the binary dir, not generated_dir, so consumers
# write #include "generated/common_generated.h". The prefix is the point: a bare
# common_generated.h looks like a checked-in header, and the one thing a reader needs to know
# about this file is that flatc writes it and editing it is pointless.
target_sources(schema INTERFACE FILE_SET HEADERS BASE_DIRS "${CMAKE_CURRENT_BINARY_DIR}" FILES ${generated_headers})
target_include_directories(schema INTERFACE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(schema INTERFACE flatbuffers::flatbuffers)

# --------------------------------------------------------------- users

# Accounts and the slimes they own. STATIC rather than INTERFACE because manager.cpp carries
# the password hashing — the implementation file the earlier comment here predicted.
#
# The rule it used to state ("depends on nothing") narrows rather than disappears: slime_users
# depends on libsodium, and PRIVATE keeps that off the LINK line of everything downstream —
# slime_world links slime_users, slime_codec links slime_world, and neither ends up linking
# libsodium.
#
# It does NOT keep sodium.h off their include path, and no CMake spelling can: every vcpkg
# package is installed into one shared prefix, so the -isystem that EnTT and spdlog already
# put on slime_codec's command line is the same directory sodium.h sits in. The rule that
# manager.h must not include <sodium.h> is therefore a convention this file cannot enforce.
# What it buys is real anyway — the crypto has exactly one translation unit, so there is one
# place to audit — but a reader should not mistake it for a wall.
add_library(slime_users STATIC "${src_dir}/users/manager.cpp")
target_include_directories(slime_users PUBLIC "${src_dir}")
target_sources(slime_users PUBLIC
    FILE_SET HEADERS
    BASE_DIRS "${src_dir}"
    FILES "${src_dir}/users/manager.h"
)
target_link_libraries(slime_users PRIVATE unofficial-sodium::sodium)
set_target_properties(slime_users PROPERTIES VERIFY_INTERFACE_HEADER_SETS TRUE)

# --------------------------------------------------------------- simulation
# Still deliberately does NOT link uwebsockets: the world must not be able to reach the
# network, and this is where that rule is enforced rather than remembered.
#
# It DOES link schema, so world.h may use the generated wire types (SlimeRepublics::Direction
# and friends) directly. That is a deliberate shortcut, not the end state — the world knowing
# the wire format is the seam this file otherwise protects. Linking it here is what makes the
# dependency real: slime_world now waits for flatc instead of reaching into a build tree by
# relative path and hoping the header is already there.

# Header-only: world.h carries the implementation too, so there is no second file to tab to
# while the simulation is in flux. INTERFACE rather than STATIC because there is nothing left
# to compile — it exists to carry the include path and the dependency edges.
add_library(slime_world INTERFACE
        "${src_dir}/game/entities.h"
        "${src_dir}/game/events.h")
# spdlog is linked here, so every consumer of slime_world can log — the simulation included.
# That is a deliberate exception to the rule two comments up: the world may not reach the
# network, but it may say what it is doing. spdlog is built WITHOUT its fmt feature (see
# vcpkg.json), so it formats through std::format and the std::formatter specializations in
# primitives.h work in a log call with no second set of fmt::formatters to maintain.
target_link_libraries(slime_world INTERFACE EnTT::EnTT schema spdlog::spdlog slime_users)
target_include_directories(slime_world INTERFACE "${src_dir}")

# The headers are listed so IDEs file them under this target rather than under whichever
# executable happened to be open when they were created. FILE_SET HEADERS (not plain
# target_sources) because these are headers, not translation units: it records them for the
# IDE and for install() without ever handing them to the compiler as sources.
#
# BASE_DIRS restates the include directory set just above. That is deliberate, not a leftover:
# FILE_SET needs it to compute each header's install-relative path, and it appends the same
# path to the interface include directories, so deleting either line still leaves the other
# working. Keep both — the explicit target_include_directories is what a reader looks for.
target_sources(slime_world INTERFACE
    FILE_SET HEADERS
    BASE_DIRS "${src_dir}"
    FILES
        "${src_dir}/game/primitives.h"
        "${src_dir}/game/world.h"
)

# Compile each header in the set on its own, as its own translation unit, and fail the build
# if one does not stand up alone. This is what catches a header that only works because some
# other header happened to include <format> first: reorder two includes and the failure would
# otherwise surface somewhere unrelated. CMake generates a verify target per header set.
set_target_properties(slime_world PROPERTIES VERIFY_INTERFACE_HEADER_SETS TRUE)

# -------------------------------------------------------------------- codec

# The wire layer: FlatBuffers in, FlatBuffers out. Links schema and slime_world because
# codec.h includes both, and deliberately does NOT link uwebsockets — the same rule the world
# lives under, for the same reason. Encoding produces bytes; deciding where they go is the
# transport's job, and not linking the socket library is what keeps that true rather than
# remembered.
#
# The first target here that actually compiles something, so it needs its own edge to flatc:
# an INTERFACE library cannot propagate an order-only dependency to its consumers.
add_library(slime_codec STATIC "${src_dir}/transport/codec.cpp"
        src/transport/formatting.h)
target_sources(slime_codec PUBLIC
    FILE_SET HEADERS
    BASE_DIRS "${src_dir}"
    FILES
        "${src_dir}/transport/codec.h"
        "${src_dir}/transport/formatting.h"
)
set_target_properties(slime_codec PROPERTIES VERIFY_INTERFACE_HEADER_SETS TRUE)
add_dependencies(slime_codec generate_schema)
target_link_libraries(slime_codec PUBLIC slime_world schema)

# ------------------------------------------------------------------- server

# codec.cpp is NOT listed here: it belongs to slime_codec, and naming it in both targets
# compiles it twice. CLion offers to add every new file to this target — decline, and put it
# in the library it belongs to instead.
add_executable(slime_server "${src_dir}/main.cpp")
# main.cpp includes common_generated.h directly, and an INTERFACE library cannot carry an
# order-only dependency to its consumers, so this target needs its own edge to flatc too.
add_dependencies(slime_server generate_schema)

# CMake excludes the header-set verification targets from "all", so a plain `cmake --build`
# would never run them. Hanging them off the executable is what makes a header that does not
# stand alone break the ordinary build rather than a target nobody remembers to invoke.
add_dependencies(slime_server all_verify_interface_header_sets)
# tomlplusplus is linked ONLY here. Reading configuration is the executable's job; nothing
# below main should be able to ask what the config file says, for the same reason the world
# does not link uwebsockets. Header-only and dependency-free, so this adds nothing to link.
target_link_libraries(slime_server PRIVATE
    tomlplusplus::tomlplusplus
    slime_codec
    slime_world
    schema
    unofficial::uwebsockets::uwebsockets
    unofficial-sodium::sodium
)

# --------------------------------------------------------------------- tests

# Links slime_codec rather than reaching for the sources directly: the tests exercise the
# same library the server does, so a test that passes says something about the shipped
# artifact. CTest learns the individual TEST_CASEs from the binary via catch_discover_tests,
# which runs it with --list-tests at build time, so `ctest` reports cases and not one target.
include(CTest)
include(Catch)

add_executable(backend_tests tests/direction_mapping_test.cpp)
add_dependencies(backend_tests generate_schema)
target_link_libraries(backend_tests PRIVATE slime_codec slime_world schema Catch2::Catch2WithMain)
catch_discover_tests(backend_tests)
